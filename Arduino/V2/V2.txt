int findTileIndex (const char* target) {
  for (int i = 0; i < BOARD_SIZE; i++) {
    if (strcasecmp(board.names[i], target) == 0) {
        return i;
    }
  }

  return -1;
};

DiceResult rollDice() {
    DiceResult d;

    uint8_t die1 = random(1, 7);
    uint8_t die2 = random(1, 7);

    d.total = die1 + die2;
    d.isDouble = (die1 == die2);

    return d;
};
